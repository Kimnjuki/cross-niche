# ── AI & search policy ────────────────────────────────────────────────────────
# We WANT to be found by search engines AND every AI assistant (Google AI
# Overviews, Gemini, ChatGPT, Claude, Perplexity, Copilot, Apple Intelligence).
# AI answer citations are a primary referral channel, so all crawlers — search,
# AI search/retrieval (ai-input), and AI training (ai-train) — are allowed.
# We publish llms.txt and llms-full.txt precisely so AI systems can read us.

Content-Signal: search=yes, ai-input=yes, ai-train=yes, use=full

# ── Private / member pages (never index) ─────────────────────────────────────
# NOTE: robots.txt patterns are PREFIX matches, not exact paths. A bare
# `Disallow: /auth` would also block every /author/<name> bio page (a core
# E-E-A-T signal submitted in the sitemap), so auth/admin/subscription/api
# rules are anchored with `$` where they have no trailing slash.
#
# IMPORTANT — do NOT add a Disallow rule for a page that already carries a
# `<meta name="robots" content="noindex">` tag. Google cannot read a noindex
# tag on a page it is not allowed to fetch, so the URL is never dropped from
# the index and instead sits in "Discovered, currently not indexed" forever.
# Every route in NON_INDEXABLE_ROUTES (scripts/lib/route-metadata.mjs) already
# emits `noindex, follow` in its static shell, so it must stay crawlable here.
# Blocking is reserved for URL patterns that should never be requested at all
# (parameters, drafts, probes) — not for pages that need a directive read.
Disallow: /admin/
Disallow: /admin$
Disallow: /billing

# ── Unpublished preview / draft routes ───────────────────────────────────────
Disallow: /preview
Disallow: /draft
Disallow: /*/preview
Disallow: /*/draft
Disallow: /*?*preview=*
Disallow: /*?*draft=*

# ── Author bio pages stay crawlable (E-E-A-T bylines) ────────────────────────
Allow: /author/

# ── Search/filter variants duplicate the canonical listings ──────────────────
# These are parameterised duplicates of canonical listings, so they are blocked
# rather than noindexed: there is no single canonical page to point at.
Disallow: /search
Disallow: /*?search=*
Disallow: /*?*&search=*
Disallow: /*?filter=*
Disallow: /*?*&filter=*

# ── Legacy / malformed article helpers ───────────────────────────────────────
Disallow: /article-detail
Disallow: /article/*/edit

# ── WordPress scanner bait (block probes only; not WordPress) ─────────────────
Disallow: /wp-admin/
Disallow: /wp-includes/
