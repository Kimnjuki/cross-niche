# AI crawlers get the SAME rules as the generic group below. The robots.txt
# spec allows multiple User-agent lines in one group, which avoids repeating
# the disallow list per bot while still naming each crawler explicitly
# (AI/GEO discoverability: GPTBot, ClaudeBot, PerplexityBot, CCBot,
# Applebot-Extended).
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: PerplexityBot
User-agent: CCBot
User-agent: Applebot-Extended
Allow: /

# Private / member pages (never index)
# NOTE (2026-09-28): robots.txt patterns are PREFIX matches, not exact paths.
# `Disallow: /auth` also blocked every /author/<name> bio page — 12 pages that
# are submitted in sitemap.xml and are a core E-E-A-T signal ("Submitted URL
# blocked by robots.txt"). Auth rules are therefore anchored with `$`. The same
# gotcha applies to /admin, /subscription and /api, which have no trailing slash.
Disallow: /admin/
Disallow: /admin$
Disallow: /profile
Disallow: /auth$
Disallow: /auth/
Disallow: /signin
Disallow: /signup
Disallow: /settings
Disallow: /billing
Disallow: /subscription$
Disallow: /subscription/
Disallow: /bookmarks
Disallow: /original-index
Disallow: /notifications
Disallow: /api$
Disallow: /api/
Disallow: /security-profile
Disallow: /preview
Disallow: /draft
Disallow: /*/preview
Disallow: /*/draft

# Author bio pages must stay crawlable — they carry the E-E-A-T bylines.
Allow: /author/

# Search/filter variants duplicate the canonical editorial listings.
Disallow: /search
Disallow: /*?*preview=*
Disallow: /*?*draft=*
Disallow: /*?search=*
Disallow: /*?*&search=*
Disallow: /*?filter=*
Disallow: /*?*&filter=*
