# ── AI & search policy ────────────────────────────────────────────────────────
# We WANT to be found by both search engines AND AI assistants (Google AI
# Overviews, ChatGPT search, Perplexity, Claude). AI answer citations are a
# primary referral channel, so all crawlers are allowed by default — search
# engines, AI search/retrieval (ai-input), and AI training (ai-train). This is
# why we publish llms.txt and llms-full.txt: so AI systems can read our content.

Content-Signal: search=yes, ai-input=yes, ai-train=yes, use=full

# All crawlers are allowed by default, including the AI crawlers:
#   Googlebot / Bingbot          — search indexing
#   OAI-SearchBot / ChatGPT-User — ChatGPT search
#   ClaudeBot / Claude-Web       — Claude
#   PerplexityBot                — Perplexity
#   GPTBot                       — OpenAI training
#   Google-Extended              — Gemini
#   CCBot                        — Common Crawl (training corpus)
#   Applebot / Applebot-Extended — Apple Search / Apple Intelligence
User-agent: *
Allow: /

# ── Private / member pages (never index) ─────────────────────────────────────
# NOTE: robots.txt patterns are PREFIX matches, not exact paths. A bare
# `Disallow: /auth` would also block every /author/<name> bio page (a core
# E-E-A-T signal submitted in the sitemap), so auth/admin/subscription/api
# rules are anchored with `$` where they have no trailing slash.
Disallow: /admin/
Disallow: /admin$
Disallow: /profile
Disallow: /auth$
Disallow: /auth/
Disallow: /signin
Disallow: /signup
Disallow: /settings
Disallow: /billing
Disallow: /subscription$
Disallow: /subscription/
Disallow: /bookmarks
Disallow: /original-index
Disallow: /notifications

# ── API responses + personal results (not editorial content) ─────────────────
Disallow: /api$
Disallow: /api/
Disallow: /security-profile

# ── Unpublished preview / draft routes ───────────────────────────────────────
Disallow: /preview
Disallow: /draft
Disallow: /*/preview
Disallow: /*/draft
Disallow: /*?*preview=*
Disallow: /*?*draft=*

# ── Author bio pages stay crawlable (E-E-A-T bylines) ────────────────────────
Allow: /author/

# ── Search/filter variants duplicate the canonical listings ──────────────────
Disallow: /search
Disallow: /*?search=*
Disallow: /*?*&search=*
Disallow: /*?filter=*
Disallow: /*?*&filter=*

# ── Dev / test pages (not for indexing) ──────────────────────────────────────
Disallow: /simple
Disallow: /enhanced
Disallow: /enhanced-simple
Disallow: /test-features
Disallow: /newsletter/verify
Disallow: /tools-old

# ── Legacy / malformed article helpers ───────────────────────────────────────
Disallow: /article-detail
Disallow: /article/*/edit

# ── WordPress scanner bait (block probes only; not WordPress) ─────────────────
Disallow: /wp-admin/
Disallow: /wp-includes/
